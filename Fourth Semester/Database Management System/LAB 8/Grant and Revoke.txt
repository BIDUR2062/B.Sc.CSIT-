--create a user name Sam and grant him the CREATE SESSION system privilege 
--so that he can log in the Oracle Database:
--Syntax: CREATE USER username DENTIFIED BY password [QUOTA {size | UNLIMITED} ON tablespace]
create user Sam identified by admin quota unlimited on users;

OR 
create user Sam identified by admin;
grant unlimited tablespace to Sam;

GRANT CREATE SESSION TO Sam;

--show the current user
SELECT sys_context('USERENV','CURRENT_USER') FROM dual;
